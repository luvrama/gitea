import os
import re
from collections import defaultdict

# Find all Go files in routers (excluding tests)
router_dir = "routers"
service_pattern = re.compile(r'"code\.gitea\.io/gitea/services/([^"]+)"')

# Map each controller file to its set of service dependencies
controller_services = {}

for root, dirs, files in os.walk(router_dir):
    for f in files:
        if f.endswith('.go') and not f.endswith('_test.go'):
            filepath = os.path.join(root, f)
            with open(filepath, 'r') as fh:
                content = fh.read()
            services = set(service_pattern.findall(content))
            if services:
                # Use relative path as controller identifier
                controller_services[filepath] = services

# Now let's group by directory (controller package) rather than individual files
# This gives a more meaningful "controller" level
package_services = defaultdict(set)
for filepath, services in controller_services.items():
    # Use the directory as the controller package
    pkg_dir = os.path.dirname(filepath)
    package_services[pkg_dir].update(services)

# Filter to packages with at least 2 services (otherwise not interesting)
package_services = {k: v for k, v in package_services.items() if len(v) >= 2}

print(f"Found {len(package_services)} controller packages with 2+ service dependencies\n")

# Print top packages by number of service deps
